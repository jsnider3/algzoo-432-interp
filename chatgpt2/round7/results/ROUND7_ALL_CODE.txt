"""Deterministic cached margin summaries; every model choice uses training folds.
Run from the extracted project: python /path/aggregates.py --recipe unif --K 1
This script never evaluates sampled inputs or accesses noisy weight replicas.
"""
import argparse, json, time
from pathlib import Path
import numpy as np
from scipy.linalg import eigh
from scipy.spatial.distance import cdist
from threadpoolctl import threadpool_limits

GAM = [.003, .01, .03, .1, .3, 1.]
ALP = np.logspace(-5, 3, 9)

def extra_features(c, K):
    """Return tails (15K columns), temporal groups (24K), and scalar-op bounds."""
    m4, m3 = c['m4'].astype(float), c['m3'].astype(float)
    s = np.sqrt((m4*m4).mean(1, keepdims=True))
    a, b = m4/s, m3/s
    i4, i3 = c['I4'], c['I3']
    tail = [v < t for v in (a, b) for t in [-.1, -.03, .01, .03, .1, .3]]
    tail += [abs(a-b), (a-b)**2, i4 != i3]
    T = np.stack(tail, -1).reshape(len(a), 720, K, -1).mean(1).reshape(len(a), -1)
    M, S, R = c['arr'].T
    masks = [M == 9, S == 9, M == S+1, M > S, R > S, abs(M-S) >= 5]
    vals = np.stack([i4, i3, a/(.1+abs(a)), b/(.1+abs(b))], -1)
    vals = vals.reshape(len(a), 720, K, 4)
    H = np.concatenate([vals[:, mask].mean(1).reshape(len(a), -1) for mask in masks], 1)
    # Baseline already computes a,b and the two softsigns. Costs include accumulation.
    ops = {'tails':32*720*K+15*K, 'groups':4*K*sum(m.sum() for m in masks)+24*K}
    return T, H, ops

def load(recipe, K):
    if recipe == 'unif':
        c = np.load(f'pop_unif/cells_k{K}.npz')
        base, y = np.c_[c['stats'], c['F']], c['acc']
        net = float(c['net_fe'])
    else:
        c = np.load(f'pop1k/seed0_z46_cells_b1_k{K}.npz')
        p = np.load('pop1k/seed0.npz'); snap = max(int(k[1:-4]) for k in p.files if k.startswith('s') and k.endswith('_Wih'))
        released = [np.load('weights.npz')]+[np.load(f'weights_seed{i}.npz') for i in range(5)]
        keys = ['rnn.weight_ih_l0','rnn.weight_hh_l0','linear.weight']
        W = [np.concatenate([p[f's{snap}_{key}'],np.stack([z[name][:,0] if j == 0 else z[name] for z in released])]).astype(float)
             for j,(key,name) in enumerate(zip(['Wih','Whh','Wo'],keys))]
        wi,A,O=W
        wf=np.stack([wi,np.diagonal(A,axis1=1,axis2=2),np.log(np.linalg.norm(A,axis=2)+1e-9),np.log(np.linalg.norm(A,axis=1)+1e-9),np.log(np.linalg.norm(O,axis=1)+1e-9)],2)
        st=np.concatenate([wf.mean(1),wf.std(1),wf.min(1),wf.max(1)],1)
        m4,m3 = c['m4'].astype(float),c['m3'].astype(float)
        s = np.sqrt((m4*m4).mean(1,keepdims=True)); a,b = m4/s,m3/s
        f = np.stack([c['I4'],c['I3'],a,b,abs(a),abs(b),a/(.1+abs(a)),b/(.1+abs(b)),a/(.03+abs(a)),b/(.03+abs(b))],-1)
        F = np.c_[s,np.log(s),m4.mean(1),m3.mean(1),f.mean(1),f.reshape(len(a),720,K,10).mean(1).reshape(len(a),-1)]
        base, y = np.c_[st,F],np.load('pop1k/reference.npz')['acc']
        net = float(np.load(f'pop_unif/cells_k{K}.npz')['net_fe'])
    T,H,ops = extra_features(c,K)
    return base,T,H,y,net,ops

def predictions(X, extras, y, tr, va, modes):
    """Ridge + RBF grids with block distances and training-only standardization."""
    def distances(Z):
        mean = Z[tr].mean(0); sd = np.maximum(Z[tr].std(0),1e-6)
        A,B = (Z[tr]-mean)/sd,(Z[va]-mean)/sd
        return A,B,cdist(A,A,'sqeuclidean')/Z.shape[1],cdist(B,A,'sqeuclidean')/Z.shape[1]
    A,B,D,V = distances(X); yt = y[tr]-y[tr].mean()
    E = {k: distances(z)[2:] for k,z in extras.items()}
    out, names = [], []
    ev,U = eigh(A.T@A); q=U.T@(A.T@yt)
    out.append(((B@U)*q)@(1/(np.maximum(ev,0)[:,None]+ALP[None]*len(tr))))
    names += [('base','ridge',0.,float(al)) for al in ALP]
    for mode,weight in modes:
        Dt,Dv = (D,V) if mode == 'base' else (D+weight*E[mode][0],V+weight*E[mode][1])
        for g in GAM:
            ev,U=eigh(np.exp(-g*Dt)); q=U.T@yt
            out.append(((np.exp(-g*Dv)@U)*q)@(1/(np.maximum(ev,0)[:,None]+ALP[None])))
            names += [(mode,float(weight),float(g),float(al)) for al in ALP]
    return y[tr].mean()+np.concatenate(out,1),names

def evaluate(X, extras, y, modes, kinds=None):
    idx=np.arange(1024); fold=np.random.default_rng(0).permutation(1024)%8
    pred={k:np.zeros(len(y)) for k in (kinds or ['base','tails','groups','both','select'])}
    picks={k:[] for k in pred}; outs=[]
    for outer in range(9):
        tr = idx if outer == 8 else idx[fold != outer]
        va = np.arange(1024,len(y)) if outer == 8 else idx[fold == outer]
        inner = np.arange(len(tr))%8 if outer == 8 else (np.arange(len(tr))*4//len(tr))[np.argsort(np.argsort(tr*7919%1024))]
        loss=None
        for j in range(inner.max()+1):
            pr,names=predictions(X,extras,y,tr[inner!=j],tr[inner==j],modes)
            err=((pr-y[tr[inner==j],None])**2).sum(0)
            loss=err if loss is None else loss+err
        pr,names=predictions(X,extras,y,tr,va,modes)
        for kind in pred:
            eligible = np.array([kind == 'select' or nm[0] in ('base',kind) for nm in names])
            choice=np.where(eligible)[0][np.argmin(loss[eligible])]
            pred[kind][va]=pr[:,choice]; picks[kind].append(names[choice])
        print('outer',outer,'choices', {k:picks[k][-1] for k in pred},flush=True)
    return pred,picks

def main():
    p=argparse.ArgumentParser(); p.add_argument('--recipe',choices=['unif','zoo'],default='unif')
    p.add_argument('--K',type=int,default=1); p.add_argument('--out',default='/workspace/scratch/0aa5efd25118/round7_work')
    a=p.parse_args(); started=time.time()
    X,T,H,y,net,ops=load(a.recipe,a.K)
    extras={'tails':T,'groups':H,'both':np.c_[T,H]}
    modes=[('base',0.)]+[(k,w) for k in extras for w in [.1,1.]]
    pred,picks=evaluate(X,extras,y,modes)
    results={}
    for kind in pred:
        d=X.shape[1]+(extras[kind].shape[1] if kind in extras else 0 if kind == 'base' else T.shape[1]+H.shape[1])
        add = 0 if kind == 'base' else ops['tails'] if kind == 'tails' else ops['groups'] if kind == 'groups' else sum(ops.values())
        # Per-block scaling adds at most d multiplies; all features charged even if baseline selected.
        cost=net+(40*720*a.K+5000+2*1024*d+34*1024+2*d+3+add+d)/2880
        groups={'crossfit':np.arange(1024)}
        groups.update({'b128':np.arange(1024,1088),'b64':np.arange(1088,1152),'mystery':np.array([1152])} if a.recipe=='unif' else {'mystery':np.array([1024]),'seeds':np.arange(1025,1030)})
        summary={}
        for name,idx in groups.items():
            error=pred[kind][idx]-y[idx]
            summary[name]={'rms_points':float(100*np.sqrt(np.mean(error**2))),'mean_points':float(100*error.mean()),'mse_ratio':float(np.mean(error**2)*cost/np.mean(y[idx]*(1-y[idx])))}
            if len(idx)<=6: summary[name]['errors_points']=(100*error).tolist()
        results[kind]={'d':d,'rows':1440*a.K,'net_fe':net,'add_feature_ops':int(add),'total_fe':cost,'summary':summary,'picks':picks[kind]}
        print(kind,json.dumps(results[kind]),flush=True)
    out=Path(a.out); out.mkdir(exist_ok=True)
    stem=out/f'aggregates_{a.recipe}_k{a.K}'
    np.savez_compressed(str(stem)+'.npz',**pred,y=y,T=T,H=H,X=X)
    Path(str(stem)+'.json').write_text(json.dumps({'recipe':a.recipe,'K':a.K,'runtime_seconds':time.time()-started,'results':results},indent=2))
    print('runtime_seconds',time.time()-started,flush=True)

if __name__ == '__main__':
    with threadpool_limits(limits=1): main()

def summarize(d, k):
    m4, m3 = d['m4'].astype(float), d['m3'].astype(float)
    s = np.sqrt(np.mean(m4*m4, axis=1, keepdims=True))
    a, b = m4/s, m3/s
    f = np.stack([d['I4'], d['I3'], a, b, abs(a), abs(b),
                  a/(.1+abs(a)), b/(.1+abs(b)),
                  a/(.03+abs(a)), b/(.03+abs(b))], axis=-1)
    return np.c_[s, np.log(s), m4.mean(1), m3.mean(1), f.mean(1),
                 f.reshape(len(f), 720, k, 10).mean(1).reshape(len(f), -1)]

def score(pred, target, cost):
    e = pred-target
    return dict(n=len(e), rms=100*np.sqrt(np.mean(e*e)), mean=100*e.mean(),
                ratio=float(np.mean(e*e)*cost/np.mean(target*(1-target))))

from threadpoolctl import threadpool_limits

GAM = [.003, .01, .03, .1, .3, 1.]
ALP = np.logspace(-5, 3, 9)
ARR = np.array(list(itertools.permutations(range(10), 3)))
FREE = [[i for i in range(10) if i not in a[:2]] for a in ARR]
RRANK = np.array([f.index(a[2]) for f, a in zip(FREE, ARR)])
PAIR = {0: 7, 7: 0, 1: 3, 3: 1, 2: 5, 5: 2, 4: 6, 6: 4}
QREP = np.array([f[PAIR[r]] for f, r in zip(FREE, RRANK)])
FOLD = np.random.default_rng(0).permutation(1024) % 8

def weights(path):
    z = np.load(path)
    if 'rnn.weight_ih_l0' in z:
        return [z['rnn.weight_ih_l0'].reshape(16).astype(float),
                z['rnn.weight_hh_l0'].astype(float),z['linear.weight'].astype(float)]
    if 'Wih' in z:
        return [z[k].astype(float) for k in ('Wih', 'Whh', 'Wo')]
    t = max(int(k[1:-4]) for k in z.files if k.startswith('s') and k.endswith('_Wih'))
    return [z[f's{t}_{k}'].astype(float) for k in ('Wih', 'Whh', 'Wo')]

def stats(wi, wh, wo):
    f = np.stack([wi, np.diagonal(wh, axis1=1, axis2=2),
                  np.log(np.linalg.norm(wh, axis=2) + 1e-9),
                  np.log(np.linalg.norm(wh, axis=1) + 1e-9),
                  np.log(np.linalg.norm(wo, axis=1) + 1e-9)], 2)
    return np.concatenate([f.mean(1), f.std(1), f.min(1), f.max(1)], 1)

def load(root, population, k):
    if population == 'unif':
        z = np.load(root / f'pop_unif/cells_k{k}.npz')
        return z, z['stats'], z['acc'].astype(float)
    z = np.load(root / f'pop1k/seed0_z46_cells_b1_k{k}.npz')
    p = weights(root / 'pop1k/seed0.npz')
    zz = [weights(root / name) for name in ['weights.npz'] + [f'weights_seed{i}.npz' for i in range(5)]]
    # Released models store the original PyTorch key names.
    w = [np.concatenate([p[j], np.stack([a[j] for a in zz])]) for j in range(3)]
    return z, stats(*w), np.load(root / 'pop1k/reference.npz')['acc'].astype(float)

def design(k, ar, kind='both'):
    fq = betaincinv(7, 4, .5)
    fr = fq + (1-fq) * betaincinv(1, 3, .5)
    fm = fr + (1-fr) * betaincinv(2, 1, .5)
    q, r, m = ndtri([fq, fr, fm])
    s = ndtri(fr + (fm-fr)*(np.arange(k)+.5)/k)
    mu = lambda v: -np.exp(-v*v/2)/np.sqrt(2*np.pi)/ndtr(v)
    out = []
    for a in ar:
        for si in s:
            for typ in ([4,3] if kind == 'both' else [int(kind)]):
                row = np.full(10, mu(q if typ == 4 else r))
                row[ARR[a]] = [m, si, r]
                if typ == 4:
                    row[QREP[a]] = q
                out.append(row)
    return np.array(out, dtype=np.float32)

def trie_plan(x):
    vals, tok = np.unique(x, return_inverse=True)
    tok = tok.reshape(x.shape)
    par, levels = np.zeros(len(x), dtype=int), []
    for t in range(10):
        key, par = np.unique(par*len(vals)+tok[:,t], return_inverse=True)
        levels.append((key//len(vals), key%len(vals)))
    return vals, levels, par

def row_cost(x):
    vals, levels, _ = trie_plan(x)
    n = np.array([len(p) for p, _ in levels])
    mac = 256*n[:-1].sum()+16*len(vals)+160*n[-1]
    other = 16*n[1:].sum()+16*n.sum()+20*n[-1]
    return float((mac+other)/2880)

def features(z, st, k, ar, kind='both'):
    ci = (ar[:,None]*k+np.arange(k)).ravel()
    if kind != 'both':
        ma, ia = z[f'm{kind}'][:,ci].astype(float), z[f'I{kind}'][:,ci]
        s = np.maximum(np.sqrt((ma*ma).mean(1,keepdims=True)),1e-15)
        a = ma/s
        f = np.stack([ia,a,abs(a),a/(.1+abs(a)),a/(.03+abs(a))],-1)
        ctx = [s,np.log(s),ma.mean(1,keepdims=True)]
    else:
        m4, m3 = [z[name][:, ci].astype(float) for name in ('m4','m3')]
        i4, i3 = [z[name][:, ci] for name in ('I4','I3')]
        s = np.maximum(np.sqrt((m4*m4).mean(1,keepdims=True)),1e-15)
        a, b = m4/s, m3/s
        f = np.stack([i4,i3,a,b,abs(a),abs(b),a/(.1+abs(a)),b/(.1+abs(b)),a/(.03+abs(a)),b/(.03+abs(b))],-1)
        ctx = [s,np.log(s),m4.mean(1,keepdims=True),m3.mean(1,keepdims=True)]
    ff = f.reshape(len(st),len(ar),k,-1)
    return np.concatenate([st]+ctx+[f.mean(1)]+[ff[:,:,i].mean(1) for i in range(k)],1)

def predictions(x, y, tr, te, supports=0):
    """Candidate predictions. Fit work is offline; compiled query is O(supports*d)."""
    mu, sd = x[tr].mean(0), np.maximum(x[tr].std(0),1e-6)
    a, b = (x[tr]-mu)/sd, (x[te]-mu)/sd
    ym = y[tr].mean()
    yc = y[tr]-ym
    ev, v = eigh(a.T@a)
    coef = v@((v.T@(a.T@yc))[:,None]/(np.maximum(ev,0)[:,None]+len(tr)*ALP))
    ans = [b@coef]
    for g in GAM:
        if supports:
            ix = np.random.default_rng(1701).permutation(len(tr))[:min(supports,len(tr))]
            centers = a[ix]
            km = np.exp(-g/x.shape[1]*cdist(centers,centers,'sqeuclidean'))
            e, u = eigh(km)
            keep = e > max(1e-12,e[-1]*1e-12)
            whitening = u[:,keep]/np.sqrt(e[keep])
            ka = np.exp(-g/x.shape[1]*cdist(a,centers,'sqeuclidean'))
            phi = ka@whitening
            e, v = eigh(phi.T@phi)
            coef = whitening@v@((v.T@(phi.T@yc))[:,None]/(np.maximum(e,0)[:,None]+ALP))
        else:
            centers = a
            ka = np.exp(-g/x.shape[1]*cdist(a,a,'sqeuclidean'))
            e, v = eigh(ka)
            coef = v@((v.T@yc)[:,None]/(np.maximum(e,0)[:,None]+ALP))
        kb = np.exp(-g/x.shape[1]*cdist(b,centers,'sqeuclidean'))
        ans.append(kb@coef)
    return ym+np.concatenate(ans,1)

def evaluate(x, y, supports=0):
    pop = np.arange(1024)
    out = np.empty(len(y)); outer, picks = [], []
    for f in range(8):
        tr, te = pop[FOLD!=f], pop[FOLD==f]
        inner = (np.arange(len(tr))*4//len(tr))[np.argsort(np.argsort(tr*7919%1024))]
        loss = sum(((predictions(x,y,tr[inner!=j],tr[inner==j],supports)-y[tr[inner==j],None])**2).sum(0) for j in range(4))
        p = predictions(x,y,tr,te,supports)
        best = int(loss.argmin()); picks.append(best)
        out[te] = p[:,best]; outer.append((te,p))
    loss = sum(((p-y[te,None])**2).sum(0) for te,p in outer)
    best = int(loss.argmin())
    out[1024:] = predictions(x,y,pop,np.arange(1024,len(y)),supports)[:,best]
    return out, picks, best

def metrics(p, y, total, population):
    groups = [('crossfit',slice(0,1024))]
    groups += [('b128',slice(1024,1088)),('b64',slice(1088,1152)),('mystery',slice(1152,1153))] if population=='unif' else [('mystery',slice(1024,1025)),('seeds',slice(1025,1030))]
    out = {}
    for name, sl in groups:
        e = p[sl]-y[sl]; mse = (e*e).mean()
        out[name] = dict(rms=100*float(np.sqrt(mse)),mean=100*float(e.mean()),ratio=float(mse*total/(y[sl]*(1-y[sl])).mean()))
        if name in ('seeds','mystery'): out[name]['errors'] = (100*e).tolist()
    return out

def main():
    root, out = Path(sys.argv[1]).resolve(), Path(sys.argv[2]).resolve()
    out.mkdir(parents=True,exist_ok=True)
    # Fixed structural designs: all ordered (M,S) pairs, with R in free ranks 0 and 7.
    designs = [(1,'ends','both',0),(2,'ends','both',0),(4,'ends','both',0),
               (2,'all','4',0),(4,'all','4',0),(4,'all','3',0),
               (1,'all','both',128),(2,'all','both',128),(4,'all','both',128)]
    if len(sys.argv)>3:
        designs = [designs[int(i)] for i in sys.argv[3].split(',')]
    for popname in ('unif','zoo'):
        for k, subset, kind, supports in designs:
            name = f'{popname}_k{k}_{subset}_mf{kind}_sup{supports or 1024}'
            if (out/f'{name}.json').exists(): continue
            start = time.monotonic()
            z, st, y = load(root,popname,k)
            ar = np.where(np.isin(RRANK,[0,7]))[0] if subset=='ends' else np.arange(720)
            x = features(z,st,k,ar,kind)
            rows = design(k,ar,kind)
            net = row_cost(rows); cells = len(ar)*k
            # Conservative feature extraction, with context, norm/log and standardization allowance.
            feature_fe = ((40 if kind=='both' else 22)*cells+5000)/2880
            readout_fe = (2*(supports or 1024)*x.shape[1]+34*(supports or 1024)+2*x.shape[1]+3)/2880
            total = net+feature_fe+readout_fe
            pred,picks,best = evaluate(x,y,supports)

                          rows=len(rows),cells=cells,dimensions=x.shape[1],net_fe=net,feature_fe=feature_fe,
                          readout_fe=readout_fe,total_fe=total,outer_picks=picks,refit_pick=best,
                          metrics=metrics(pred,y,total,popname),seconds=time.monotonic()-start)
            np.savez_compressed(out/f'{name}.npz',pred=pred,y=y,X=x)
            (out/f'{name}.json').write_text(json.dumps(result,indent=2))

"""Compiled inference for the Round 7 estimators (NumPy/SciPy only).

Offline: select candidate hyperparameters on training networks, then call
compile_readout. Online: fixed_prefix_logits and predict use target weights
or features only. This implements the arithmetic assumed by the FE ledger.
"""
import numpy as np
from scipy.linalg import eigh
from scipy.spatial.distance import cdist
from cost_experiments import ALP, GAM, trie_plan

def fixed_prefix_logits(wi, wh, wo, plan):
    """One network; plan=trie_plan(fixed_rows) is prepared offline.

    Input products are computed once per distinct input value. Recurrent
    products are computed once per unique parent state. No sampled inputs.
    """
    values, levels, leaves = plan
    inp = values[:,None]*np.asarray(wi).reshape(1,-1)
    h = None
    for parent, token in levels:
        h = np.maximum(inp[token],0) if h is None else np.maximum((h@wh.T)[parent]+inp[token],0)
    return (h@wo.T)[leaves]

def cell_outputs(wi, wh, wo, plan, target):
    lg = fixed_prefix_logits(wi,wh,wo,plan)
    correct = lg.argmax(1) == target  # NumPy chooses the lowest tied index.
    chosen = lg[np.arange(len(lg)),target].copy()
    lg[np.arange(len(lg)),target] = -np.inf
    return correct, chosen-lg.max(1)

def compile_readout(x, y, choice, supports=0):
    """Offline fit; choice is column number in cost_experiments.predictions.

    Exact RBF uses every training support; supports>0 uses a fixed,
    training-only inducing subset and a Nystrom kernel approximation.
    No validation or target rows enter this function.
    """
    mu, sd = x.mean(0), np.maximum(x.std(0),1e-6)
    a = (x-mu)/sd
    ym = y.mean(); yc=y-ym
    family, penalty = divmod(int(choice),len(ALP)); alpha=ALP[penalty]
    if family == 0:
        e,v=eigh(a.T@a)
        beta=v@((v.T@(a.T@yc))/(np.maximum(e,0)+len(y)*alpha))
        coef=beta/sd
        return dict(family='linear',coef=coef,intercept=ym-mu@coef)
    gamma=GAM[family-1]/x.shape[1]
    centers=a
    if supports:
        idx=np.random.default_rng(1701).permutation(len(y))[:min(supports,len(y))]
        centers=a[idx]
        km=np.exp(-gamma*cdist(centers,centers,'sqeuclidean'))
        e,v=eigh(km); keep=e>max(1e-12,e[-1]*1e-12)
        whitening=v[:,keep]/np.sqrt(e[keep])
        phi=np.exp(-gamma*cdist(a,centers,'sqeuclidean'))@whitening
        e,v=eigh(phi.T@phi)
        coef=whitening@v@((v.T@(phi.T@yc))/(np.maximum(e,0)+alpha))
    else:
        e,v=eigh(np.exp(-gamma*cdist(a,a,'sqeuclidean')))
        coef=v@((v.T@yc)/(np.maximum(e,0)+alpha))
    return dict(family='rbf',mu=mu,sd=sd,centers=centers,
                coef=coef,intercept=ym,gamma=gamma)

def predict(model, x):
    """Online inference; one selected compiled readout, no fitting/eigensystem."""
    if model['family']=='linear':
        return np.asarray(x)@model['coef']+model['intercept']
    a=np.atleast_2d((x-model['mu'])/model['sd'])
    distance=cdist(a,model['centers'],'sqeuclidean')
    return model['intercept']+np.exp(-model['gamma']*distance)@model['coef']

def mf4_cells(w, k):
    rows=design(k,np.arange(720),'4');vals,levels,leaf=trie_plan(rows)
    target=np.repeat(ARR[:,1],k);bits=[];margins=[]
    for start in range(0,len(w[0]),24):
        wi,wh,wo=[a[start:start+24].astype(np.float32) for a in w]
        inp=wi[:,None,:]*vals[None,:,None];h=None
        for parent,token in levels:
            h=np.maximum(inp[:,token],0) if h is None else np.maximum((h@wh.transpose(0,2,1))[:,parent]+inp[:,token],0)
        lg=(h@wo.transpose(0,2,1))[:,leaf]
        bits.append(lg.argmax(2)==target[None])
        chosen=lg[:,np.arange(len(target)),target].copy()
        lg[:,np.arange(len(target)),target]=-np.inf
        margins.append(chosen-lg.max(2))
    return {'I4':np.concatenate(bits),'m4':np.concatenate(margins)}

def temporal4(z,k):
    m=z['m4'].astype(float);s=np.maximum(np.sqrt((m*m).mean(1,keepdims=True)),1e-15)
    a=m/s;v=np.stack([z['I4'],a/(.1+abs(a))],-1).reshape(len(m),720,k,2)
    M,S,R=ARR.T
    masks=[M==9,S==9,M==S+1,M>S,R>S,abs(M-S)>=5]
    h=np.concatenate([v[:,mask].mean(1).reshape(len(m),-1) for mask in masks],1)
    # If called separately from features, include normalizing and softsign again.
    ops=6*720*k+2*k*sum(int(mask.sum()) for mask in masks)+12*k
    return h,ops

def weights(recipe):
    names=['pop_unif/b128_n1024_seed4.npz','pop_unif/b128_seed0.npz','pop_unif/b64_seed5.npz'] if recipe=='unif' else ['pop1k/seed0.npz']

def selection(k):
    ids=[]
    for i,(m,s,r) in enumerate(G.ARR):
        free=[t for t in range(10) if t not in (m,s)]
        if (m==9 or m==s+1) and r in (free[0],free[-1]): ids.append(i)
    slices=[0] if k==1 else [0,k-1]
    return np.array([i*k+j for i in ids for j in slices]),np.array(slices)

def margin_grad(w,x,y):
    """Signed target-versus-best-rival margin and the active ReLU Jacobian."""
    wi,A,O=w; n=len(wi); h=np.zeros((n,len(x),16)); gates=[]
    for t in range(10):
        pre=x[None,:,t,None]*wi[:,None,:]
        if t: pre=pre+h@A.transpose(0,2,1)
        gates.append(pre>0); h=np.maximum(pre,0)
    logits=h@O.transpose(0,2,1)
    target=logits[:,np.arange(len(x)),y].copy()
    logits[:,np.arange(len(x)),y]=-np.inf
    rival=logits.argmax(2)  # NumPy argmax preserves the lowest-index tie rule.
    margin=target-np.take_along_axis(logits,rival[...,None],2)[...,0]
    adj=O[:,y,:]-O[np.arange(n)[:,None],rival,:]
    grad=np.zeros((n,len(x),10))
    for t in range(9,-1,-1):
        adj=adj*gates[t]; grad[:,:,t]=np.sum(adj*wi[:,None,:],axis=2)
        if t: adj=adj@A
    return margin,grad

def extract(w,k,s):
    """Differentiates actual anchor gates and active output rival; eight features/slice."""
    ids,sl=selection(k); x=G.rows(1,k)[2*ids].astype(float)
    margin,grad=margin_grad(w,x,G.ARR[ids//k,1])
    lower=x<0; grad*=lower[None]
    q=float(np.sort(x[0])[-4]); mu=float(G.mu(q))
    sigma=np.sqrt(1+q*mu-mu*mu)
    z=grad*(sigma/np.maximum(s,1e-12))[:,None,None]
    l2=np.sqrt(np.sum(z*z,axis=2)); a=margin/np.maximum(s,1e-12)[:,None]
    pair=lower[:,:-1]&lower[:,1:]
    both=(z[:,:,:-1]+z[:,:,1:])*pair[None]
    diff=(z[:,:,:-1]-z[:,:,1:])*pair[None]
    f=np.stack([l2,np.sum(abs(z),axis=2)/np.sqrt(6),
                abs(z.sum(2))/np.sqrt(6),np.sqrt(np.sum(z[:,:,8:]**2,axis=2)),
                np.sqrt(np.sum(both*both,axis=2)/5),
                np.sqrt(np.sum(diff*diff,axis=2)/5),
                (a>0)&(a<=l2),l2/(abs(a)+l2+1e-12)],axis=2)
    out=np.concatenate([f[:,ids%k==j,:].mean(1) for j in sl],axis=1)
    return out,dict(rows=len(x),dims=out.shape[1],backward_fe=len(x)*2640/2880,
                    summary_fe=(len(x)*160+out.shape[1])/2880,
                    # Includes explicit gate comparisons in this NumPy version.
                    standalone_forward_fe=len(x)*3108/2880,
                    q=q,sigma=sigma)

"""Nested learning curve for fixed MF4 + pulses, K=2, 128 supports.

Run next to cost_experiments.py:
  python learning_curve.py PATH/TO/unif_k2_all_mf4_p1_sup128.npz
The adjacent JSON supplies the constant deployed cost. No test labels enter
subset selection, standardization, hyperparameter choice, or fitting.
"""
import json, sys, time
from pathlib import Path
import numpy as np
from threadpoolctl import threadpool_limits
from cost_experiments import predictions, FOLD

def score(p, y, cost):
    e=p-y; mse=float(np.mean(e*e))
    return dict(rms=100*np.sqrt(mse),mean=100*float(e.mean()),
                ratio=mse*cost/float(np.mean(y*(1-y))))

def main(path):
    src=Path(path); d=np.load(src); x,y=d['X'],d['y']
    cost=json.loads(src.with_suffix('.json').read_text())['total_fe']
    pop=np.arange(1024); external=np.arange(1024,len(y))
    priority=np.random.default_rng(202607).permutation(1024)
    records=[]; allpred=[]; started=time.monotonic()
    for nt,nfinal in zip((224,448,672,896),(256,512,768,1024)):
        tic=time.monotonic(); cf=np.empty(1024); picks=[]; outer=[]
        for f in range(8):
            eligible=priority[FOLD[priority]!=f]
            tr=np.sort(eligible[:nt]); te=pop[FOLD==f]
            inner=(np.arange(nt)*4//nt)[np.argsort(np.argsort(tr*7919%1024))]
            loss=sum(((predictions(x,y,tr[inner!=j],tr[inner==j],128)
                       -y[tr[inner==j],None])**2).sum(0) for j in range(4))
            best=int(loss.argmin()); candidates=predictions(x,y,tr,te,128)
            cf[te]=candidates[:,best]; picks.append(best); outer.append((te,candidates))
        # Fixed nested subsets of the 1024 population. Refit parameters come
        # from eight training-only folds on that subset, preserving FOLD IDs.
        tr=np.sort(priority[:nfinal])
        if nfinal==1024:
            loss=sum(((p-y[te,None])**2).sum(0) for te,p in outer)
        else:
            loss=sum(((predictions(x,y,tr[FOLD[tr]!=f],tr[FOLD[tr]==f],128)
                       -y[tr[FOLD[tr]==f],None])**2).sum(0) for f in range(8))
        best=int(loss.argmin()); extra=predictions(x,y,tr,external,128)[:,best]
        pp=np.r_[cf,extra]; allpred.append(pp)
        r=dict(outer_train_count=nt,external_train_count=nfinal,total_fe=cost,
               crossfit=score(cf,y[:1024],cost),outer_picks=picks,refit_pick=best,
               b128=score(pp[1024:1088],y[1024:1088],cost),
               b64=score(pp[1088:1152],y[1088:1152],cost),
               mystery_error=100*float(pp[-1]-y[-1]),seconds=time.monotonic()-tic)
        records.append(r); print(json.dumps(r),flush=True)
    out=Path(__file__).parent
    result=dict(method='MF4 + pulses, K=2, 128 supports',source=str(src),
                priority_seed=202607,outer_folds='rng0 permutation(1024) % 8',
                inner_folds='4, stable IDs and denominator1024',
                supports=128,features=x.shape[1],total_fe=cost,records=records,
                total_seconds=time.monotonic()-started)
    (out/'learning_curve.json').write_text(json.dumps(result,indent=2))
    np.savez_compressed(out/'learning_curve.npz',pred=np.array(allpred),y=y,
                        priority=priority,outer_train=[224,448,672,896],
                        external_train=[256,512,768,1024])

if __name__=='__main__':
    with threadpool_limits(limits=1): main(sys.argv[1])

def impulse_features(wi,A,O):
    """16 basis-pointer summaries plus 10 signed input-pulse summaries.

    Logits are normalized independently by age and injected direction; all
    summaries survive positive diagonal hidden-unit and global logit scaling.
    Basis impulses test a possible memory state, not its reachability.
    """
    n = len(wi)
    h = np.concatenate([np.broadcast_to(np.eye(16),(n,16,16)),
                        np.maximum(wi,0)[:,:,None],np.maximum(-wi,0)[:,:,None]],2)
    L = []
    for age in range(10):
        L.append(O @ h)
        if age != 9: h = np.maximum(A @ h,0)
    L = np.stack(L,1)                           # network, age, class, impulse
    lc = L-L.mean(2,keepdims=True)
    amp = np.sqrt((lc*lc).mean(2))
    scale = np.maximum(amp,1e-12)
    out = []
    allmean = []
    for lag in (0,1):
        a = np.arange(10-lag); target = 9-a-lag
        valid = L[:,a].copy(); correct = valid.argmax(2)==target[None,:,None]
        tgt = valid[:,a,target,:].copy()
        valid[:,a,target,:] = -np.inf
        margin = (tgt-valid.max(2))/scale[:,a]
        mean = margin[:,:,:16].mean(1)
        rate = correct[:,:,:16].mean(1)
        allmean.append(mean)
        out += [mean.max(1),np.sort(mean,axis=1)[:,-4:].mean(1),
                rate.max(1),np.sort(rate,axis=1)[:,-4:].mean(1),
                margin[:,:,:16].min(1).max(1),
                margin[:,5:,:16].mean(1).max(1)]
        if lag == 0:
            pulse = [mean for mean in
                     (margin[:,:,16:].mean(1),correct[:,:,16:].mean(1),
                      margin[:,5:,16:].mean(1))]
    # Choose a basis pointer using only that network's mean normalized margins.
    best = np.stack(allmean,1).reshape(n,-1).argmax(1)%16
    growth = np.diff(np.log(np.maximum(amp,1e-12)),axis=1)
    selected = growth[np.arange(n),:,best]
    out += [selected.mean(1),selected.std(1),
            growth[:,:,:16].mean((1,2)),growth[:,:,:16].std((1,2))]
    pulse += [growth[:,:,16:].mean(1),growth[:,:,16:].std(1)]
    return np.stack(out,1),np.concatenate(pulse,1)

def pulse_features(wi,A,O):
    """10 features, only two deterministic signed pulses; budget 6 FE.

    For each sign: mean age margin, age alignment fraction, late-age margin,
    and mean/SD of successive log output-amplitude changes. Age alignment
    tests how the memory decodes a pulse, not second-argmax task correctness.
    """
    h=np.maximum(np.stack([wi,-wi],2),0); ll=[]
    for age in range(10):
        ll.append(O@h)
        if age<9: h=np.maximum(A@h,0)
    L=np.stack(ll,1); lc=L-L.mean(2,keepdims=True)
    amp=np.maximum(np.sqrt((lc*lc).mean(2)),1e-12)

    target=L[:,age,tgt,:].copy(); L[:,age,tgt,:]=-np.inf
    margin=(target-L.max(2))/amp
    growth=np.diff(np.log(amp),axis=1)
    return np.concatenate([margin.mean(1),aligned.mean(1),margin[:,5:].mean(1),

def fold_preds(x,y,tr,va):
    mu=x[tr].mean(0); sd=np.maximum(x[tr].std(0),1e-6)
    a,b=(x[tr]-mu)/sd,(x[va]-mu)/sd; ym=y[tr].mean(); yt=y[tr]-ym

    out=[((b@V)*q)@(1/(np.maximum(ev,0)[:,None]+ALP[None]*len(tr)))]
    D,Dv=cdist(a,a,'sqeuclidean'),cdist(b,a,'sqeuclidean')
    for g in GAM:
        ev,V=eigh(np.exp(-g/x.shape[1]*D)); q=V.T@yt
        out.append(((np.exp(-g/x.shape[1]*Dv)@V)*q)@
                   (1/(np.maximum(ev,0)[:,None]+ALP[None])))

def nested(x,y):
    idx=np.arange(1024); folds=np.random.default_rng(0).permutation(1024)%8
    p=np.zeros(1024); pick=[]
    for f in range(8):
        tr,va=idx[folds!=f],idx[folds==f]
        inner=(np.arange(len(tr))*4//len(tr))[np.argsort(np.argsort(tr*7919%1024))]
        loss=sum(((fold_preds(x,y,tr[inner!=j],tr[inner==j])-
                   y[tr[inner==j],None])**2).sum(0) for j in range(4))
        j=int(loss.argmin()); pick.append(j)
        p[va]=fold_preds(x,y,tr,va)[:,j]
    # Refit hyperparameters chosen entirely on the 1024 training networks.
    loss=sum(((fold_preds(x,y,idx[idx%8!=f],idx[idx%8==f])-
               y[idx[idx%8==f],None])**2).sum(0) for f in range(8))
    j=int(loss.argmin())
    extra=fold_preds(x,y,idx,np.arange(1024,len(y)))[:,j]
    return np.r_[p,extra],pick,j

def make_plan(spec, ar):

def prefix_cells(w, plan, dtype=np.float64):
    """Batched targets; input products cached once per distinct scalar."""
    wi, wh, wo = [np.asarray(a, dtype=dtype) for a in w]
    inp = wi[:, None, :]*plan['values'].astype(dtype)[None, :, None]
    h = None
    for t in range(10):
        par, tok = plan[f'parent{t}'], plan[f'token{t}']
        h = np.maximum(inp[:, tok], 0) if h is None else np.maximum((h@wh.transpose(0, 2, 1))[:, par]+inp[:, tok], 0)
    lg = (h@wo.transpose(0, 2, 1))[:, plan['leaves']]

    chosen = lg[:, np.arange(len(target)), target].copy()
    lg[:, np.arange(len(target)), target] = -np.inf

    b, m = bits.reshape(n, -1, nr), margins.reshape(n, -1, nr).astype(float)
    s = np.maximum(np.sqrt((m[:, :, 0]**2).mean(1, keepdims=True)), 1e-15)
    a = m/s[:, :, None]
    aa = abs(a); soft = a/(.1+aa)
    f = np.concatenate([b, a, aa, soft, a/(.03+aa)], 2)
    base = np.c_[stats, s, np.log(s), m.mean(1), f.mean(1),
                 f.reshape(n, len(ar), k, -1).mean(1).reshape(n, -1)]
    if pulse is not None:
        base = np.c_[base, pulse]
    if spec['groups'] == 'none':
        return base, base.shape[1]
    masks = temporal_masks(ar) if masks is None else masks
    v = np.concatenate([b, soft], 2).reshape(n, len(ar), k, 2*nr)
    groups = np.concatenate([v[:, mask].mean(1).reshape(n, -1) for mask in masks], 1)
    return np.c_[base, groups], base.shape[1]


def cached_features(root, spec, cells_path=None, generate=False, cache_out=None):
    w, y = population(root, spec['recipe'])
    ar, k = arrangement_ids(spec), spec['K']
    default = (root/f'pop_unif/cells_k{k}.npz' if spec['recipe'] == 'unif'
               else root/f'pop1k/seed0_z46_cells_b1_k{k}.npz')
    path = Path(cells_path) if cells_path else default
    types = [4, 3] if spec['kind'] == 'both' else [int(spec['kind'])]
    if path.exists():
        z = dict(np.load(path, allow_pickle=False))
    else:
        if not generate:
            raise FileNotFoundError(f'{path}; use --cells PATH or --generate')
        plan = make_plan(spec, np.arange(720))
        rows = [prefix_cells([a[i:i+24] for a in w], plan, np.float32)
                for i in range(0, len(y), 24)]
        b, m = [np.concatenate([r[j] for r in rows]) for j in range(2)]
        z = {f'{prefix}{typ}': value[:, j::len(types)] for j, typ in enumerate(types)
             for prefix, value in [('I', b), ('m', m)]}
        np.savez_compressed(cache_out, **z)
    for typ in types:
        for prefix in ('I', 'm'):
            if z[f'{prefix}{typ}'].shape != (len(y), 720*k):
                raise ValueError(f'Wrong recipe/K or incomplete cache: {prefix}{typ} has shape {z[f"{prefix}{typ}"].shape}')
    ci = (ar[:, None]*k+np.arange(k)).ravel()
    bits = np.stack([z[f'I{t}'][:, ci] for t in types], 2).reshape(len(y), -1)
    margins = np.stack([z[f'm{t}'][:, ci] for t in types], 2).reshape(len(y), -1)
    p = pulse_features(*w) if spec['pulse'] else None
    x, db = summarize(bits, margins, weight_stats(*w), spec, ar, p)
    return x, y, db, make_plan(spec, ar)


def candidates(spec):
    out = [('linear', 0., 0., float(a)) for a in ALP]
    weights = [0., .1, 1.] if spec['groups'] == 'tune' else [0.]
    return out+ [('rbf', w, g, float(a)) for w in weights for g in GAM for a in ALP]


def metric(a, b, db, mode, weight):
    if mode != 'tune':
        return cdist(a, b, 'sqeuclidean')/a.shape[1]
    return (cdist(a[:, :db], b[:, :db], 'sqeuclidean')/db
            + weight*cdist(a[:, db:], b[:, db:], 'sqeuclidean')/(a.shape[1]-db))


def kernel_fit(a, y, db, spec, weight, gamma):
    """Offline eigensystems; Nystrom uses only a fixed subset of training rows."""
    ids = (np.random.default_rng(1701).permutation(len(a))[:min(spec['supports'], len(a))]
           if spec['supports'] else np.arange(len(a)))
    centers = a[ids]
    km = np.exp(-gamma*metric(centers, centers, db, spec['groups'], weight))
    e, v = eigh(km)
    if not spec['supports']:
        return centers, v@((v.T@y)[:, None]/(np.maximum(e, 0)[:, None]+ALP))
    keep = e > max(1e-12, e[-1]*1e-12)
    whitening = v[:, keep]/np.sqrt(e[keep])
    phi = np.exp(-gamma*metric(a, centers, db, spec['groups'], weight))@whitening
    e, v = eigh(phi.T@phi)
    coef = whitening@v@((v.T@(phi.T@y))[:, None]/(np.maximum(e, 0)[:, None]+ALP))
    return centers, coef


def candidate_predictions(x, y, tr, te, db, spec):
    mean, sd = x[tr].mean(0), np.maximum(x[tr].std(0), 1e-6)
    a, b = (x[tr]-mean)/sd, (x[te]-mean)/sd
    ym = y[tr].mean(); yc = y[tr]-ym
    dl = db if spec['groups'] == 'tune' else x.shape[1]
    e, v = eigh(a[:, :dl].T@a[:, :dl])
    coef = v@((v.T@(a[:, :dl].T@yc))[:, None]/(np.maximum(e, 0)[:, None]+len(tr)*ALP))
    out = [b[:, :dl]@coef]
    for weight in ([0., .1, 1.] if spec['groups'] == 'tune' else [0.]):
        for gamma in GAM:
            centers, coef = kernel_fit(a, yc, db, spec, weight, gamma)
            out.append(np.exp(-gamma*metric(b, centers, db, spec['groups'], weight))@coef)
    return ym+np.concatenate(out, 1)


def crossfit(x, y, db, spec, refit):
    pop = np.arange(1024); pred = np.empty(len(y)); picks = []; outer_loss = 0
    for f in range(8):
        tr, te = pop[FOLD != f], pop[FOLD == f]
        inner = (np.arange(len(tr))*4//len(tr))[np.argsort(np.argsort(tr*7919 % 1024))]
        loss = sum(((candidate_predictions(x, y, tr[inner != j], tr[inner == j], db, spec)
                     - y[tr[inner == j], None])**2).sum(0) for j in range(4))
        pr = candidate_predictions(x, y, tr, te, db, spec)
        pick = int(loss.argmin()); pred[te] = pr[:, pick]; picks.append(pick)
        outer_loss = outer_loss+((pr-y[te, None])**2).sum(0)
    if refit == 'index':
        outer_loss = sum(((candidate_predictions(x, y, pop[pop % 8 != j], pop[pop % 8 == j], db, spec)
                           - y[pop[pop % 8 == j], None])**2).sum(0) for j in range(8))
    choice = int(outer_loss.argmin())
    pred[1024:] = candidate_predictions(x, y, pop, np.arange(1024, len(y)), db, spec)[:, choice]
    return pred, picks, choice


def compile_model(x, y, db, spec, choice):
    """One selected readout is compiled; query-time prediction never fits."""
    family, weight, gamma, alpha = candidates(spec)[choice]
    mean, sd = x.mean(0), np.maximum(x.std(0), 1e-6)
    a, yc, ym = (x-mean)/sd, y-y.mean(), y.mean()
    if family == 'linear':
        dl = db if spec['groups'] == 'tune' else x.shape[1]
        e, v = eigh(a[:, :dl].T@a[:, :dl])
        beta = v@((v.T@(a[:, :dl].T@yc))/(np.maximum(e, 0)+len(y)*alpha))
        coef = np.r_[beta/sd[:dl], np.zeros(x.shape[1]-dl)]
        return {'family': np.array(family), 'coef': coef, 'intercept': ym-mean@coef}
    centers, coef = kernel_fit(a, yc, db, spec, weight, gamma)
    coord = np.full(x.shape[1], 1/np.sqrt(x.shape[1]))
    if spec['groups'] == 'tune':
        coord = np.r_[np.full(db, 1/np.sqrt(db)), np.full(x.shape[1]-db, np.sqrt(weight/(x.shape[1]-db)))]
    return {'family': np.array(family), 'mean': mean, 'scale': coord/sd,
            'centers': centers*coord, 'coef': coef[:, int(np.argmin(abs(ALP-alpha)))],
            'gamma': gamma, 'intercept': ym}


def predict_features(model, x):
    if str(model['family']) == 'linear':
        return np.asarray(x)@model['coef']+model['intercept']
    a = (np.atleast_2d(x)-model['mean'])*model['scale']
    return model['intercept']+np.exp(-model['gamma']*cdist(a, model['centers'], 'sqeuclidean'))@model['coef']


def predict_weights(model, weights):
    """Actual deterministic target query: cached prefix inputs + one readout."""
    spec = json.loads(str(model['spec']))
    shapes = [(16,), (16, 16), (10, 16)]
    if len(weights) != 3 or any(np.asarray(w).shape != shape for w, shape in zip(weights, shapes)):
        raise ValueError('One target requires weight shapes (16,), (16,16), (10,16)')
    w = [np.asarray(v)[None] for v in weights]
    b, m = prefix_cells(w, model)
    p = pulse_features(*w) if spec['pulse'] else None
    masks = [model[f'group{i}'] for i in range(6)]
    x, _ = summarize(b, m, weight_stats(*w), spec, model['ar'], p, masks)
    return float(predict_features(model, x)[0])


def costs(spec, plan, d, db, model):
    n = np.array([len(plan[f'parent{t}']) for t in range(10)])
    net = (256*n[:-1].sum()+16*len(plan['values'])+160*n[-1]
           +16*n[1:].sum()+16*n.sum()+20*n[-1])/2880
    k, ar = spec['K'], plan['ar']; cells = k*len(ar)
    ops = (40 if spec['kind'] == 'both' else 22)*cells+5000
    if spec['groups'] != 'none':
        masks = [plan[f'group{i}'] for i in range(6)]
        channels = 4 if spec['kind'] == 'both' else 2
        ops += channels*k*sum(int(m.sum()) for m in masks)+channels*6*k
    feature = ops/2880; pulse = 6. if spec['pulse'] else 0.
    ns = spec['supports'] or 1024
    budget = (2*ns*d+34*ns+2*d+3)/2880
    actual = (d+1)/2880 if str(model['family']) == 'linear' else budget
    return dict(anchor_rows=len(plan['target']), pulse_rows=2*int(spec['pulse']),
                prefix_fe=float(net), feature_fe=feature, pulse_fe=pulse,
                readout_budget_fe=budget, compiled_readout_fe=actual,
                total_fe=net+feature+pulse+budget, compiled_total_fe=net+feature+pulse+actual)


def metrics(pred, y, cost, recipe):
    groups = {'crossfit': slice(0, 1024)}
    groups.update({'b128': slice(1024, 1088), 'b64': slice(1088, 1152), 'mystery': slice(1152, 1153)}
                  if recipe == 'unif' else {'mystery': slice(1024, 1025), 'seeds': slice(1025, 1030)})
    out = {}
    for name, sl in groups.items():
        e = pred[sl]-y[sl]; mse = np.mean(e*e)
        out[name] = dict(rms_points=float(100*np.sqrt(mse)), mean_points=float(100*e.mean()),
                         mse_ratio=float(mse*cost/np.mean(y[sl]*(1-y[sl]))))
        if name in ('mystery', 'seeds'):
            out[name]['errors_points'] = (100*e).tolist()
    return out


def main():
    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    ap.add_argument('--root', type=Path, default=Path('.')); ap.add_argument('--out', type=Path, default=Path('round7_result'))
    ap.add_argument('--recipe', choices=['unif', 'zoo'], default='unif'); ap.add_argument('--K', type=int, default=2)
    ap.add_argument('--kind', choices=['both', '4', '3'], default='4'); ap.add_argument('--subset', choices=['all', 'ends'], default='all')
    ap.add_argument('--pulse', action=argparse.BooleanOptionalAction, default=True)
    ap.add_argument('--groups', choices=['none', 'fixed', 'tune'], default='tune'); ap.add_argument('--supports', type=int, choices=[0, 128], default=0)
    ap.add_argument('--refit', choices=['outer', 'index'], default='index'); ap.add_argument('--threads', type=int, default=1)
    ap.add_argument('--cells', type=Path); ap.add_argument('--generate', action='store_true')
    ap.add_argument('--predict', type=Path); ap.add_argument('--weights', type=Path, default=Path('weights.npz'))
    args = ap.parse_args()
    with threadpool_limits(limits=args.threads):
        if args.predict:
            with np.load(args.predict, allow_pickle=False) as model:
                print(json.dumps({'estimated_accuracy': predict_weights(model, read_weights(args.weights))}))
            return
        if args.K < 1:
            ap.error('--K must be positive')
        spec = {k: getattr(args, k) for k in ['recipe', 'K', 'kind', 'subset', 'pulse', 'groups', 'supports']}
        args.out.parent.mkdir(parents=True, exist_ok=True); started = time.monotonic()
        x, y, db, plan = cached_features(args.root, spec, args.cells, args.generate, args.out.with_suffix('.cells.npz'))
        pred, picks, choice = crossfit(x, y, db, spec, args.refit)
        model = compile_model(x[:1024], y[:1024], db, spec, choice)
        # Concrete compilation check against the selected candidate's external predictions.
        discrepancy = float(np.max(abs(predict_features(model, x[1024:])-pred[1024:])))
        if discrepancy > 1e-8:
            raise ArithmeticError(f'Compiled readout mismatch: {discrepancy}')
        ledger = costs(spec, plan, x.shape[1], db, model)
        result = dict(spec=spec, refit=args.refit, dimensions=x.shape[1], base_dimensions=db,
                      costs=ledger, metrics=metrics(pred, y, ledger['total_fe'], args.recipe),
                      outer_choices=[candidates(spec)[i] for i in picks], refit_choice=candidates(spec)[choice],
                      compile_max_error=discrepancy, offline_seconds=time.monotonic()-started)
        np.savez_compressed(args.out.with_suffix('.model.npz'), **model, **plan, spec=np.array(json.dumps(spec)))
        np.savez_compressed(args.out.with_suffix('.predictions.npz'), pred=pred, y=y, X=x)
        args.out.with_suffix('.json').write_text(json.dumps(result, indent=2))
        print(json.dumps(result, indent=2))


if __name__ == '__main__':
    main()
